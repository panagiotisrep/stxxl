# Runs and parameters for experiments/run_all.sh with FAIR=1 (equal resources, I/O model).
# See experiments/competitors/README.md, "Fair comparison", for why each value is chosen.
#
# One run per line:   <run name>  <binary>  <options ...>
#   run name   any name; it names the log files (<name>.txt, <name>.out) and can be selected with RUNS=
#   binary     boa (= BOA_BIN), or a path relative to COMP_BUILD (the build_all.sh output)
#   options    everything passed to the binary, except the workload (--size, --queries, --key-dist,
#              --query-dist, identical for every run, set by SIZE/QUERIES/KEY_DIST/QUERY_DISTS) and --log-file
# Placeholders {...} in options are computed by the script (integer arithmetic) from:
#   M       memory budget in bytes = max BOA "Max internal memory bytes" + one block (measured from the
#           boa runs, which therefore come first; every non-boa run uses M and runs under the memory limit)
#   BLOCK   I/O block size (BLOCK, default 32768 = BOA_BLOCK_SIZE)
#   SIZE    number of keys,  QUERIES  queries per distribution
#   BATCH_INSERTS, BATCH_QUERIES  insertions / queries per batch of the interleaved runs
#   {M_GIB} is M in GiB with 6 decimals (LeanStore takes GiB).
#   Example: {M/2}, {BLOCK}, {M/4 > 65536 ? M/4 : 65536}
# Text after # is a comment.
#
# Dynamic structures (BOA default mode, LevelDB, RocksDB, LeanStore, STXXL map/unordered_map, SILT combi)
# run interleaved: --interleaved, batches of BATCH_INSERTS insertions each followed by BATCH_QUERIES
# queries to keys inserted so far (QUERY_DISTS does not apply to them). Static structures (BOA
# online-suspended, PaCHash, SILT sorted store) insert everything first, then run the queries of every
# distribution in QUERY_DISTS.

# BOA: no --buffer-size, so the buffer (BOA's memory budget) is the binary's default, which depends on
# the block size: BOA_BLOCK_SIZE / run element size (13 bytes for boa.h, 16 for boa_pv.h), i.e. 2520
# and 2048 for 32 KiB blocks, 5041 and 4096 for 64 KiB.
# The block size is the unit BOA reads and writes on disk (BOA_BLOCK_SIZE, compile time; set BLOCK to
# the same value). BOA_BLOCK_SIZE, BOA_LAMBDA, BOA_PAGES need a rebuild, see boa.md.
boa                  boa   --routing-filter-mult 1 --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
boa_pv               boa   --boa-pv --routing-filter-mult 1 --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
boa_suspended        boa   --suspended --materialization hybrid --routing-filter-mult 1
boa_pv_suspended     boa   --boa-pv --suspended --materialization hybrid --routing-filter-mult 1
# Online-suspended and interleaved: consolidate_structure() after every batch (compare with rocksdb_bulk_il).
# Other materializations: copy a line and change --materialization (merge, sort, hybrid).
boa_suspended_il     boa   --suspended --materialization hybrid --routing-filter-mult 1 --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
boa_pv_suspended_il  boa   --boa-pv --suspended --materialization hybrid --routing-filter-mult 1 --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}

# LevelDB: memtable + immutable memtable = M/2, block cache = M/2. No Bloom filter: LevelDB keeps filters
# outside its cache (unbounded), so a filter would be memory beyond M. No direct I/O in LevelDB.
# Full compaction before the queries (driver default; timed as "Compaction").
leveldb              leveldb/leveldb_bench   --write-buffer-size {M/4 > 65536 ? M/4 : 65536} --cache-size {M/2} --block-size {BLOCK} --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
# LevelDB with batched writes (--write-batch): 1000 keys per leveldb::WriteBatch, i.e. one DB::Write, one
# log record and one memtable insertion step per 1000 keys; the rest of every interleaved batch is
# written before its queries. Compaction still runs automatically (LevelDB cannot postpone it).
# 1000 keys keep the memtable overshoot small (it may exceed write_buffer_size by one WriteBatch).
leveldb_batch        leveldb/leveldb_bench   --write-buffer-size {M/4 > 65536 ? M/4 : 65536} --cache-size {M/2} --block-size {BLOCK} --write-batch 1000 --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
#   leveldb_batch       interleaved (compare with boa, rocksdb)
#   leveldb_batch_then  insert-then-query, all query distributions (compare with boa_suspended, rocksdb_bulk)
leveldb_batch_then   leveldb/leveldb_bench   --write-buffer-size {M/4 > 65536 ? M/4 : 65536} --cache-size {M/2} --block-size {BLOCK} --write-batch 1000

# RocksDB: 2 memtables = M/2, block cache = M/2 including index and filter blocks (so a Bloom filter
# lives inside M), direct I/O for reads, flushes and compactions, one background job.
# Buffers outside the cache cut to one block: compaction read-ahead per input file (RocksDB default 2 MiB)
# and write buffer per file being written (default 1 MiB); with the defaults RocksDB used 21 MB (77 MB in
# bulk mode) above the key set at 300k keys, with one block each 6.5 MB (8 MB).
# Without --bloom-bits: no filter (with 10 bits/key the filters do not fit in M/2 at 2M keys).
rocksdb              rocksdb/rocksdb_bench   --write-buffer-size {M/4} --cache-size {M/2} --cache-index-and-filter-blocks --block-size {BLOCK} --direct-reads --direct-writes --max-background-jobs 1 --compaction-readahead-size {BLOCK} --writable-file-max-buffer-size {BLOCK} --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
rocksdb_bloom        rocksdb/rocksdb_bench   --write-buffer-size {M/4} --cache-size {M/2} --cache-index-and-filter-blocks --bloom-bits 10 --block-size {BLOCK} --direct-reads --direct-writes --max-background-jobs 1 --compaction-readahead-size {BLOCK} --writable-file-max-buffer-size {BLOCK} --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
# RocksDB bulk load (--bulk-load), its counterpart of BOA's online-suspended mode: automatic compaction
# off, insertions only fill memtables and L0 files, one full compaction (CompactRange) consolidates.
#   rocksdb_bulk     insert-then-query, one compaction after all insertions (compare with boa_suspended)
#   rocksdb_bulk_il  interleaved, a compaction after every batch, counted as insertion time
#                    (compare with boa_suspended_il)
rocksdb_bulk         rocksdb/rocksdb_bench   --write-buffer-size {M/4} --cache-size {M/2} --cache-index-and-filter-blocks --block-size {BLOCK} --direct-reads --direct-writes --max-background-jobs 1 --compaction-readahead-size {BLOCK} --writable-file-max-buffer-size {BLOCK} --bulk-load
rocksdb_bulk_il      rocksdb/rocksdb_bench   --write-buffer-size {M/4} --cache-size {M/2} --cache-index-and-filter-blocks --block-size {BLOCK} --direct-reads --direct-writes --max-background-jobs 1 --compaction-readahead-size {BLOCK} --writable-file-max-buffer-size {BLOCK} --bulk-load --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}

# LeanStore: buffer pool = M; plain B-tree, no transactions/WAL (BOA has neither); one worker thread.
# Page size fixed at 4 KiB. Buffers outside the pool cut down: WAL buffer (allocated even without WAL,
# default 10 MiB) to one block, page-provider write buffer from 1024 to 32 pages, 2 instead of 64
# buffer partitions; this took LeanStore from 44 MB to 19 MB above the key set (the rest is fixed).
leanstore            leanstore/leanstore_bench   --dram_gib={M_GIB} --worker_threads=1 --pp_threads=1 --vi=false --isolation_level=ru --wal=false --wal_buffer_size={BLOCK} --write_buffer_size=32 --partition_bits=1 --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}

# STXXL B+-tree: node cache + leaf cache = M. Node/leaf block size is compile time (32 KiB).
stxxl_map            stxxl/stxxl_map_bench   --node-cache {M/2} --leaf-cache {M/2} --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}

# STXXL unordered_map: insertion buffer = M/2, block cache = M/2 (stxxl::hash_map::tuning, in blocks; STXXL's
# default is 12 blocks), prefetching 1 page of 1 block (default 2 x 2 blocks). Built with 1 sub-block per
# block (build_all.sh), so its block, the unit it caches and reads, is BLOCK.
# Buckets: STXXL's own rehash target N / (entries per sub-block * 0.875);
# the map never rehashes by itself. Entry = 8 bytes (uint32_t key + char value, padded).
stxxl_unordered_map  stxxl/stxxl_unordered_map_bench   --buffer-size {M/2} --block-cache-blocks {M/2/BLOCK > 0 ? M/2/BLOCK : 1} --prefetch-page-size 1 --prefetch-pages 1 --initial-buckets {(SIZE * 64 + BLOCK * 7 - 1) / (BLOCK * 7)} --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}

# SILT: its memory use is not configurable (SILT's own configuration, direct I/O).
silt_combi           silt/silt_bench   --mode combi --interleaved --insertions-per-batch {BATCH_INSERTS} --queries-per-batch {BATCH_QUERIES}
silt_sorted          silt/silt_bench   --mode sorted

# PaCHash (static): defaults, O_DIRECT, posix I/O, queue depth 1; block length is compile time (= BLOCK).
pachash              pachash/pachash_bench   --a 8 --io posix --queue-depth 1
