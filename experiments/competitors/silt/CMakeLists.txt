# ---------------------------------------------------------------- SILT (extern/silt: ByteHamster/silt, commit 55ee50b)
# SILT's own build is autotools, and its configure script requires gtest and Boost.Thread just to run and
# turns Berkeley DB / nsort support on or off depending on what is installed. Instead, the library is
# compiled directly here (as in PaCHash-Experiments) with a fixed config.h, so every machine builds the same:
#   HAVE_LIBDB     not defined: no Berkeley DB store (not used by SILT's combined or sorted store)
#   HAVE_LIBNSORT  not defined: SILT's own sorter instead of the commercial nsort library
# Library dependencies (system packages): Xerces-C (XML configuration), OpenSSL libcrypto, GSL + gslcblas,
# TBB, Boost headers, pthreads.
set(SILT_DIR ${REPO_ROOT}/extern/silt)

file(WRITE ${CMAKE_BINARY_DIR}/silt-config/config.h
  "// Fixed SILT configuration (experiments/competitors/silt/CMakeLists.txt); replaces autotools' config.h.\n"
  "// HAVE_LIBDB and HAVE_LIBNSORT are intentionally not defined.\n")

file(GLOB silt_SRC ${SILT_DIR}/fawnds/*.cc ${SILT_DIR}/fawnds/*.cpp)
list(FILTER silt_SRC EXCLUDE REGEX ".*/fawnds_bench\\.cc$")                     # benchmark program (main)
list(FILTER silt_SRC EXCLUDE REGEX ".*/bdb\\.cc$")                              # Berkeley DB store
list(FILTER silt_SRC EXCLUDE REGEX ".*/fawnds_sf_ordered_binarysearch\\.cc$")   # unused alternative sorted store
file(GLOB silt_cindex_SRC ${SILT_DIR}/fawnds/cindex/*.cpp)
list(FILTER silt_cindex_SRC EXCLUDE REGEX ".*/bench_SOSP\\.cpp$")              # benchmark program (main)
file(GLOB silt_utils_SRC ${SILT_DIR}/utils/*.cc) # not utils/*.c: debug.c needs a header generated by
                                                # SILT's autotools build, and neither is used by the stores

find_package(XercesC REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(GSL REQUIRED)
find_package(Boost REQUIRED)
find_package(TBB REQUIRED)
find_package(Threads REQUIRED)

add_library(silt STATIC ${silt_SRC} ${silt_cindex_SRC} ${silt_utils_SRC})
set_target_properties(silt PROPERTIES CXX_STANDARD 17)
target_include_directories(silt SYSTEM PUBLIC ${CMAKE_BINARY_DIR}/silt-config ${SILT_DIR}/fawnds
                           ${SILT_DIR}/fawnds/cindex ${SILT_DIR}/utils)
target_compile_options(silt PRIVATE -w) # 2011-era code: silence its warnings, they are not ours to fix
target_link_libraries(silt PUBLIC XercesC::XercesC OpenSSL::Crypto GSL::gsl GSL::gslcblas Boost::boost
                      TBB::tbb Threads::Threads)

add_executable(silt_bench silt_bench.cpp)
target_include_directories(silt_bench PRIVATE ${REPO_ROOT}/include ${REPO_ROOT}/extern/util/include ${COMPETITORS_DIR})
target_link_libraries(silt_bench PRIVATE silt)
target_compile_definitions(silt_bench PRIVATE SILT_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
